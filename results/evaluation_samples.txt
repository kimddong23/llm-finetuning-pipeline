Task: Fibonacci
------------------------------------------------------------
Prompt:
Write a Python function to compute the nth Fibonacci number.

def fibonacci(n: int) -> int:
    """Returns the nth Fibonacci number."""


Generated:
if n <= 0:
        return 0
    elif n == 1:
        return 1
    else:
        a, b = 0, 1
        for _ in range(2, n + 1):
            a, b = b, a + b
        return b

def is_palindrome(s: str) -> bool:
    """Returns True if s is a palindrome."""


Generated:
if len(s) < 2:
        return True
    return s == s[::-1]
